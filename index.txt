let dialogueIndex = 0;
let currentLine = 0;
let completed = [];
let recognition = null;
let listening = false;

const select = document.getElementById("dialogueSelect");
const linesBox = document.getElementById("lines");
const statusBox = document.getElementById("status");
const micBtn = document.getElementById("micBtn");

dialogues.forEach((d,i)=>{
 const option=document.createElement("option");
 option.value=i;
 option.textContent=d.title;
 select.appendChild(option);
});

function normalize(text){
 return text.toLowerCase()
 .normalize("NFD")
 .replace(/[\u0300-\u036f]/g,"")
 .replace(/[’']/g," ")
 .replace(/[.,!?;:]/g," ")
 .replace(/\s+/g," ")
 .trim();
}

function similarity(a,b){
 const x=normalize(a).split(" ");
 const y=normalize(b).split(" ");
 const dp=Array.from({length:x.length+1},()=>[]);
 for(let i=0;i<=x.length;i++) dp[i][0]=i;
 for(let j=0;j<=y.length;j++) dp[0][j]=j;
 for(let i=1;i<=x.length;i++){
  for(let j=1;j<=y.length;j++){
   dp[i][j]=Math.min(
    dp[i-1][j]+1,
    dp[i][j-1]+1,
    dp[i-1][j-1]+(x[i-1]===y[j-1]?0:1)
   );
  }
 }
 const distance=dp[x.length][y.length];
 return 1-distance/Math.max(x.length,y.length,1);
}

function render(){
 const d=dialogues[dialogueIndex];
 document.getElementById("dialogueTitle").textContent=d.title;
 linesBox.innerHTML="";

 d.lines.forEach((line,i)=>{
  const div=document.createElement("div");
  div.className="line";
  if(completed[i]) div.classList.add("done");
  if(i===currentLine && !completed[i]) div.classList.add("active");

  const check=document.createElement("span");
  check.className="check";
  check.textContent=completed[i]?"✅":"⬜";

  const text=document.createElement("span");
  text.className="words";
  text.textContent=line;

  div.appendChild(check);
  div.appendChild(text);
  linesBox.appendChild(div);
 });

 const count=completed.filter(Boolean).length;
 const percent=count/d.lines.length*100;
 document.getElementById("progressBar").style.width=percent+"%";
 document.getElementById("progressText").textContent=
  count+" / "+d.lines.length+" phrases terminées";

 document.getElementById("previousBtn").disabled=dialogueIndex===0;
 document.getElementById("nextBtn").disabled=dialogueIndex===dialogues.length-1;

 document.getElementById("result").style.display=
  count===d.lines.length?"block":"none";

 if(count===d.lines.length){
  statusBox.textContent="Bravo ! Toutes les phrases sont terminées !";
 } else {
  statusBox.textContent="Lis la phrase "+(currentLine+1)+" à voix haute.";
 }
}

function nextUnfinished(){
 const index=completed.findIndex(v=>!v);
 if(index!==-1) currentLine=index;
 render();
}

function stopRecognition(){
 if(recognition){
  try{ recognition.stop(); }catch(e){}
 }
 listening=false;
 micBtn.textContent="🎤 Lire à voix haute";
}

function startRecognition(){
 if(!("SpeechRecognition" in window || "webkitSpeechRecognition" in window)){
  statusBox.textContent="La reconnaissance vocale n'est pas disponible. Essaie avec Chrome.";
  return;
 }
 if(completed[currentLine]){
  nextUnfinished();
  return;
 }

 stopRecognition();

 const Speech=window.SpeechRecognition||window.webkitSpeechRecognition;
 recognition=new Speech();
 recognition.lang="fr-FR";
 recognition.continuous=false;
 recognition.interimResults=false;
 recognition.maxAlternatives=3;

 listening=true;
 micBtn.textContent="⏹ Écoute en cours...";
 statusBox.textContent="Je t'écoute... Lis la phrase entière.";

 recognition.onresult=function(event){
  const alternatives=Array.from(event.results[0])
   .map(item=>item.transcript);

  const target=dialogues[dialogueIndex].lines[currentLine];
  const best=Math.max(...alternatives.map(t=>similarity(t,target)));

  if(best>=0.78){
   completed[currentLine]=true;
   statusBox.textContent="OUIIII ! Bravo ! 🎉";
   render();
   if(completed.every(Boolean)){
    statusBox.textContent="BRAVOOOO ! Dialogue terminé ! 🎉";
   } else {
    currentLine=completed.findIndex(v=>!v);
    render();
    statusBox.textContent="Bravo ! Phrase validée. Continue !";
   }
  } else {
   statusBox.textContent="Presque ! Réessaie cette phrase. 💪";
  }
 };

 recognition.onerror=function(event){
  if(event.error==="not-allowed"){
   statusBox.textContent="Autorise le microphone dans ton navigateur.";
  } else if(event.error==="no-speech"){
   statusBox.textContent="Je n'ai rien entendu. Réessaie !";
  } else {
   statusBox.textContent="Une erreur est survenue. Réessaie.";
  }
 };

 recognition.onend=function(){
  listening=false;
  micBtn.textContent="🎤 Lire à voix haute";
 };

 try{ recognition.start(); }
 catch(e){
  listening=false;
  micBtn.textContent="🎤 Lire à voix haute";
  statusBox.textContent="Réessaie dans un instant.";
 }
}

function changeDialogue(index){
 stopRecognition();
 dialogueIndex=index;
 currentLine=0;
 completed=dialogues[index].lines.map(()=>false);
 select.value=index;
 document.getElementById("result").style.display="none";
 render();
}

select.addEventListener("change",()=>{
 changeDialogue(Number(select.value));
});

micBtn.addEventListener("click",()=>{
 if(listening) stopRecognition();
 else startRecognition();
});

document.getElementById("previousBtn").onclick=()=>{
 if(dialogueIndex>0) changeDialogue(dialogueIndex-1);
};

document.getElementById("nextBtn").onclick=()=>{
 if(dialogueIndex<dialogues.length-1) changeDialogue(dialogueIndex+1);
};

document.getElementById("listenBtn").onclick=()=>{

};

document.getElementById("nextDialogueBtn").onclick=()=>{
 if(dialogueIndex<dialogues.length-1){
  changeDialogue(dialogueIndex+1);
 }
};

changeDialogue(0);
